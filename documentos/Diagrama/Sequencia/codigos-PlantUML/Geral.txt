title Diagrama de Sequência Geral - Sistema UniRewards Completo

actor "Administrador\n(Instituição)" as admin
actor "Empresa\nParceira" as empresa
actor "Professor" as prof
actor "Aluno" as aluno

participant "API UniRewards" as api
database "Banco de Dados" as db
queue "Fila (RabbitMQ)" as mq

' =========================================================================
== MÓDULO 1: GESTÃO DE INSTITUIÇÃO ==
' =========================================================================
autonumber 1.1
admin -> api: POST /api/instituicoes (Cadastrar)
activate api
api -> db: SELECT verifica se sigla existe
db --> api: Retorna Falso
api -> api: Codifica senha (BCrypt) e define tipoUsuario="INSTITUICAO"
api -> db: INSERT nova instituição
db --> api: Retorna Instituicao salva
api --> admin: 201 Created
deactivate api

autonumber 1.2
admin -> api: GET, PUT ou DELETE /api/instituicoes/{id}
activate api
api -> db: Realiza operação (SELECT, UPDATE ou DELETE)
db --> api: Retorna resultado
api --> admin: 200 OK / 204 No Content
deactivate api

' =========================================================================
== MÓDULO 2: GESTÃO DE PROFESSORES ==
' =========================================================================
autonumber 2.1
admin -> api: POST /api/professores (Cadastrar Individual)
activate api

db --> api: Retorna Falso
api -> api: Codifica senha e define tipoUsuario="PROFESSOR"
api -> db: INSERT novo professor
db --> api: Retorna Professor salvo
api -> db: INSERT nova carteira com 1000 Lúmens
db --> api: Retorna Carteira salva
api --> admin: 201 Created
deactivate api

autonumber 2.2
prof -> api: GET, PUT ou DELETE /api/professores/{id}
activate api
api -> db: Realiza operação (SELECT, UPDATE ou DELETE)
db --> api: Retorna resultado
api --> prof: 200 OK / 204 No Content
deactivate api

' =========================================================================
== MÓDULO 5: TRANSAÇÕES (DISTRIBUIÇÃO DE LÚMENS) ==
' =========================================================================
autonumber 5.1
prof -> api: POST /api/transacoes (Professor para Aluno)
activate api
api -> db: SELECT busca Aluno Destinatário, Professor Remetente e Carteiras
db --> api: Retorna entidades e carteiras

alt Saldo Suficiente
    api -> api: Subtrai valor do Professor e adiciona ao Aluno
    api -> db: UPDATE carteiras
    db --> api: Confirmação
    api -> db: INSERT log (transacaoProfessorRepository.save)
    db --> api: Retorna log salvo
    
    api -> mq: Envia payload de e-mails para fila
    activate mq
    mq --> api: Confirmação de enfileiramento
    
    api --> prof: 200 OK ("Transferência processada com sucesso!")
    
    mq -> aluno: E-mail assíncrono (RECEBIDO)
    mq -> prof: E-mail assíncrono (ENVIADO)
    deactivate mq
else Saldo Insuficiente
    api --> prof: 400 Bad Request
end
deactivate api

autonumber 5.2
aluno -> api: POST /api/transacoes (Aluno para Aluno)
activate api
api -> api: Verifica se Remetente != Destinatário
api -> db: SELECT busca Alunos e Carteiras
db --> api: Retorna entidades e carteiras

alt Saldo Suficiente
    api -> api: Subtrai valor do Remetente e adiciona ao Destinatário
    api -> db: UPDATE carteiras
    db --> api: Confirmação
    api -> db: INSERT log (transacaoAlunoRepository.save)
    db --> api: Retorna log salvo
    
    api -> mq: Envia payload de e-mails para fila
    activate mq
    mq --> api: Confirmação de enfileiramento
    
    api --> aluno: 200 OK ("Transferência processada com sucesso!")
    
    mq -> aluno: E-mail assíncrono (RECEBIDO)
    mq -> aluno: E-mail assíncrono (ENVIADO)
    deactivate mq
else Saldo Insuficiente ou IDs Iguais
    api --> aluno: 400 Bad Request
end
deactivate api

' =========================================================================
== MÓDULO 6: LOJA E RESGATE DE VANTAGENS ==
' =========================================================================
autonumber 6.1
aluno -> api: GET /api/vantagens (Visualizar Loja)
activate api
api -> db: SELECT busca todas as vantagens ativas
db --> api: Retorna lista
api --> aluno: 200 OK (Catálogo)
deactivate api

autonumber 6.2
aluno -> api: POST /api/resgates (Resgatar Vantagem)
activate api
api -> db: SELECT busca Aluno, Carteira e Vantagem
db --> api: Retorna dados (com custo da vantagem)

alt Saldo Suficiente
    api -> api: Subtrai custo da Carteira do Aluno
    api -> db: UPDATE Carteira e UPDATE Empresa (totalResgatados++)
    db --> api: Confirmação de atualizações
    api -> api: Gera código único e payload do QR Code para o Cupom
    api -> db: INSERT log na tabela de resgates
    db --> api: Retorna Resgate salvo
    
    api -> mq: Envia payload de e-mails para fila
    activate mq
    mq --> api: Confirmação de enfileiramento
    
    api --> aluno: 200 OK ("Resgate realizado com sucesso!")
    
    mq -> aluno: E-mail assíncrono (Cupom / QR Code)
    mq -> empresa: E-mail assíncrono (Aviso de Resgate)
    deactivate mq
else Saldo Insuficiente
    api --> aluno: 400 Bad Request
end
deactivate api
